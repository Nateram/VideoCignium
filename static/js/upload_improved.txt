        // Subir cada archivo en chunks SECUENCIALMENTE
        console.log('\n' + '='.repeat(80));
        console.log('📤 PASO 2: SUBIR ARCHIVOS POR FRAGMENTOS');
        console.log('='.repeat(80));
        
        for (let fileIndex = 0; fileIndex < selectedFiles.length; fileIndex++) {
            const file = selectedFiles[fileIndex];
            const totalChunks = Math.ceil(file.size / CHUNK_SIZE);
            
            console.log(`\n${'▼'.repeat(40)}`);
            console.log(`📄 PROCESANDO ARCHIVO ${fileIndex + 1}/${selectedFiles.length}`);
            console.log(`   Nombre: ${file.name}`);
            console.log(`   Tamaño: ${(file.size / (1024*1024)).toFixed(2)} MB`);
            console.log(`   Fragmentos: ${totalChunks}`);
            console.log(`${'▼'.repeat(40)}\n`);
            
            uploadProgressText.textContent = `Subiendo ${file.name} (${fileIndex + 1}/${selectedFiles.length})`;
            
            // Subir cada chunk SECUENCIALMENTE (esperando respuesta antes del siguiente)
            for (let chunkIndex = 0; chunkIndex < totalChunks; chunkIndex++) {
                const start = chunkIndex * CHUNK_SIZE;
                const end = Math.min(start + CHUNK_SIZE, file.size);
                const chunk = file.slice(start, end);
                
                console.log(`\n┌─ FRAGMENTO ${chunkIndex + 1}/${totalChunks} ─────────────────`);
                console.log(`│ Tamaño: ${(chunk.size / (1024*1024)).toFixed(2)} MB (${chunk.size} bytes)`);
                console.log(`│ Rango: ${start} - ${end}`);
                console.log(`│ Timestamp inicio: ${new Date().toISOString()}`);
                
                const formData = new FormData();
                formData.append('chunk', chunk);
                formData.append('filename', file.name);
                formData.append('chunkIndex', chunkIndex);
                formData.append('totalChunks', totalChunks);
                formData.append('folderId', folderId);
                
                const chunkUrl = buildApiUrl('/api/upload/chunk');
                console.log(`│ URL: ${chunkUrl}`);
                console.log(`└─────────────────────────────────────────`);
                
                // Enviar chunk y ESPERAR respuesta
                console.log(`⏳ Enviando fragmento...`);
                const uploadStartTime = Date.now();
                
                let chunkResponse = null;
                
                // Intentar hasta 3 veces
                for (let attempt = 1; attempt <= 3; attempt++) {
                    try {
                        if (attempt > 1) {
                            console.log(`\n🔄 REINTENTO ${attempt}/3 para fragmento ${chunkIndex + 1}/${totalChunks}`);
                            await new Promise(resolve => setTimeout(resolve, 2000));
                        }
                        
                        const controller = new AbortController();
                        const timeoutId = setTimeout(() => {
                            console.error(`⏰ TIMEOUT después de 5 minutos`);
                            controller.abort();
                        }, 300000);
                        
                        chunkResponse = await fetch(chunkUrl, {
                            method: 'POST',
                            body: formData,
                            signal: controller.signal
                        });
                        
                        clearTimeout(timeoutId);
                        
                        const uploadDuration = ((Date.now() - uploadStartTime) / 1000).toFixed(2);
                        const uploadSpeed = (chunk.size / (1024 * 1024) / uploadDuration).toFixed(2);
                        
                        console.log(`\n✅ RESPUESTA RECIBIDA:`);
                        console.log(`   Status: ${chunkResponse.status} ${chunkResponse.statusText}`);
                        console.log(`   Tiempo: ${uploadDuration}s`);
                        console.log(`   Velocidad: ${uploadSpeed} MB/s`);
                        console.log(`   Timestamp fin: ${new Date().toISOString()}`);
                        
                        if (!chunkResponse.ok) {
                            const errorText = await chunkResponse.text();
                            console.error(`\n❌ ERROR HTTP ${chunkResponse.status}:`);
                            console.error(`   ${errorText}`);
                            throw new Error(`HTTP ${chunkResponse.status}: ${chunkResponse.statusText}`);
                        }
                        
                        // Éxito - salir del bucle de reintentos
                        break;
                        
                    } catch (fetchError) {
                        console.error(`\n❌ ERROR en intento ${attempt}/3:`);
                        console.error(`   Tipo: ${fetchError.name}`);
                        console.error(`   Mensaje: ${fetchError.message}`);
                        
                        if (attempt === 3) {
                            console.error(`\n💥 FALLO DEFINITIVO después de 3 intentos`);
                            throw new Error(`No se pudo subir fragmento ${chunkIndex + 1}/${totalChunks}: ${fetchError.message}`);
                        }
                    }
                }
                
                if (!chunkResponse) {
                    throw new Error(`No se recibió respuesta para fragmento ${chunkIndex + 1}/${totalChunks}`);
                }
                
                // Parsear respuesta JSON
                console.log(`\n📥 Parseando respuesta JSON...`);
                const chunkData = await chunkResponse.json();
                console.log(`✅ Datos recibidos:`, chunkData);
                
                if (!chunkData.success) {
                    console.error(`\n❌ El servidor reportó error:`, chunkData);
                    throw new Error(chunkData.error || 'Error desconocido del servidor');
                }
                
                // Si es el último chunk, guardar info del archivo
                if (chunkData.complete) {
                    console.log(`\n🎉 ARCHIVO COMPLETO ENSAMBLADO EN EL SERVIDOR`);
                    console.log(`   Ruta: ${chunkData.filepath}`);
                    uploadedFiles.push({
                        name: file.name,
                        path: chunkData.filepath,
                        is_duplicate: false
                    });
                }
                
                // Actualizar progreso
                uploadedSize += (end - start);
                const percentComplete = Math.round((uploadedSize / totalSize) * 100);
                uploadProgressBar.style.width = percentComplete + '%';
                uploadProgressText.textContent = `Subiendo ${file.name}: ${percentComplete}%`;
                
                console.log(`\n📊 PROGRESO GLOBAL: ${percentComplete}%`);
                console.log(`   Subido: ${(uploadedSize / (1024*1024)).toFixed(2)} MB`);
                console.log(`   Total: ${(totalSize / (1024*1024)).toFixed(2)} MB`);
                console.log(`   Restante: ${((totalSize - uploadedSize) / (1024*1024)).toFixed(2)} MB`);
                
                // Pequeña pausa entre chunks para no saturar
                if (chunkIndex < totalChunks - 1) {
                    console.log(`\n⏸️  Pausa de 100ms antes del siguiente fragmento...`);
                    await new Promise(resolve => setTimeout(resolve, 100));
                }
            }
            
            console.log(`\n${'▲'.repeat(40)}`);
            console.log(`✅ ARCHIVO COMPLETADO: ${file.name}`);
            console.log(`${'▲'.repeat(40)}`);
        }
