server {
    server_name natera.dev www.natera.dev;

    # Escuchar en el puerto 80 (para Cloudflare Flexible)
    listen 80;

    # 🔥 CONFIGURACIÓN PARA UPLOADS GRANDES (GLOBAL)
    client_max_body_size 2000M;  # Permite archivos hasta 2GB
    client_body_timeout 600s;     # Timeout para recibir el body completo
    client_body_buffer_size 128k; # Buffer para uploads

    # --- Bloque 1: Dominio Principal (Tu Portfolio en la RPi) ---
    location / {
        root /var/www/html;
        index index.html index.htm;
        try_files $uri $uri/ =404;
    }
    
    # --- Bloque 2: App de IA (Tu PC) ---
    # 🔥 IMPORTANTE: El trailing slash en proxy_pass reescribe la ruta
    location /proyect1/ {
        proxy_pass http://192.168.1.131:5000/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # 🔥 Timeouts EXTENDIDOS para uploads de videos grandes
        proxy_connect_timeout 600s;
        proxy_send_timeout 600s;
        proxy_read_timeout 600s;

        # 🔥 CRÍTICO para uploads: deshabilitar buffering
        proxy_request_buffering off;
        proxy_buffering off;
        
        # 🔥 AÑADIR ESTAS DOS LÍNEAS (necesarias para chunking)
        proxy_http_version 1.1;
        proxy_set_header Connection "";

        # 🔥 Límites específicos para esta ruta (uploads)
        client_max_body_size 2000M;
        client_body_timeout 600s;
    }

    # --- Bloque 3: Recursos Estáticos (Solución al CSS/JS) ---
    location /static/ {
        proxy_pass http://192.168.1.131:5000/static/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }

    # 🔥 CRÍTICO: Ruta para videos subidos (necesaria para reproducción)
    location /uploads/ {
        proxy_pass http://192.168.1.131:5000/uploads/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;

        # Timeouts para videos grandes
        proxy_connect_timeout 600s;
        proxy_send_timeout 600s;
        proxy_read_timeout 600s;
        proxy_buffering off;
    }
    
    # 🔥 CRÍTICO: Ruta para videos temporales convertidos (ROI config)
    location /temp/ {
        proxy_pass http://192.168.1.131:5000/temp/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;

        # Timeouts para videos
        proxy_connect_timeout 600s;
        proxy_send_timeout 600s;
        proxy_read_timeout 600s;
        proxy_buffering off;
    }
    
    # 🔥 CRÍTICO: Ruta para clips generados (análisis de movimiento)
    location /clips/ {
        proxy_pass http://192.168.1.131:5000/clips/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;

        # Timeouts para videos
        proxy_connect_timeout 600s;
        proxy_send_timeout 600s;
        proxy_read_timeout 600s;
        proxy_buffering off;
    }

    location /css/ {
        proxy_pass http://192.168.1.131:5000/css/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }

    location /js/ {
        proxy_pass http://192.168.1.131:5000/js/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        
        # 🔥 AÑADIR ESTA LÍNEA (evita caché de JavaScript viejo)
        add_header Cache-Control "no-cache, no-store, must-revalidate";
    }

    location /webfonts/ {
        proxy_pass http://192.168.1.131:5000/webfonts/;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }

    # --- Bloque 4: WebSocket para 3 en Raya ---
    location /tictactoe-ws {
        proxy_pass http://127.0.0.1:8765;
        proxy_http_version 1.1;
        
        # Headers críticos para WebSocket
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        
        # Timeouts largos para mantener conexión WebSocket
        proxy_connect_timeout 7d;
        proxy_send_timeout 7d;
        proxy_read_timeout 7d;
        
        # Deshabilitar buffering para WebSocket
        proxy_buffering off;
    }
}
